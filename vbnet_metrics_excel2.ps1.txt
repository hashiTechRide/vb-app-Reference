#Requires -Version 7.0
<#
.SYNOPSIS
    VB.NET 2008 CE プロジェクト メトリクス解析 → Excel出力（COM経由）
.DESCRIPTION
    対象: Visual Studio 2008 Compact Edition (VB.NET) プロジェクト
    出力: Excel ブック (.xlsx) ※ Excel インストール必須
    依存: なし（PowerShell 7.x + Excel COM のみ）
.PARAMETER Path
    解析対象のプロジェクトフォルダパス
.PARAMETER OutFile
    出力 Excel ファイルパス（省略時は自動命名）
.PARAMETER NoOpen
    出力後に Excel を開かない
.EXAMPLE
    .\vbnet_metrics_excel.ps1 -Path "C:\Projects\PickingApp"
    .\vbnet_metrics_excel.ps1 -Path "C:\Projects\PickingApp" -OutFile "C:\report.xlsx"
    .\vbnet_metrics_excel.ps1 "C:\Projects\PickingApp" -NoOpen
#>

[CmdletBinding()]
param(
    [Parameter(Mandatory, Position = 0)]
    [ValidateScript({ Test-Path $_ -PathType Container })]
    [string]$Path,

    [Parameter(Position = 1)]
    [string]$OutFile,

    [switch]$NoOpen
)

$ErrorActionPreference = 'Stop'

# ============================================================
# データクラス
# ============================================================

class MethodInfo {
    [string]$Name; [string]$FilePath; [string]$ClassName
    [string]$AccessModifier; [string]$ReturnType; [string]$Params
    [int]$StartLine; [int]$EndLine; [int]$LineCount = 0
    [int]$BranchCount = 0; [int]$MaxNestDepth = 0; [int]$CyclomaticComplexity = 1
    [System.Collections.Generic.List[string]]$CallsTo = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$GVRead = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$GVWrite = [System.Collections.Generic.List[string]]::new()
    [bool]$HasDBAccess = $false; [bool]$HasNetAccess = $false
}

class GlobalVarInfo {
    [string]$Name; [string]$VarType; [string]$DeclaredIn; [string]$Modifier
    [string]$InitialValue = ''
    [System.Collections.Generic.List[string]]$SetLocations = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$ReadLocations = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.HashSet[string]]$RefFunctions = [System.Collections.Generic.HashSet[string]]::new()
    [System.Collections.Generic.HashSet[string]]$RefFiles = [System.Collections.Generic.HashSet[string]]::new()
}

class FormInfo {
    [string]$Name; [string]$FilePath
    [System.Collections.Generic.List[string]]$Controls = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[hashtable]]$DynamicControls = [System.Collections.Generic.List[hashtable]]::new()
    [System.Collections.Generic.List[string]]$EventHandlers = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$NavigatesTo = [System.Collections.Generic.List[string]]::new()
}

class BranchInfo {
    [string]$FilePath; [string]$MethodName; [int]$LineNumber
    [string]$BranchType; [string]$Condition; [string]$Target = ''; [int]$NestDepth
}

class FileMetrics {
    [string]$FilePath; [string]$FileName; [string]$FileType
    [int]$TotalLines = 0; [int]$CodeLines = 0; [int]$CommentLines = 0; [int]$BlankLines = 0
    [int]$ClassCount = 0; [int]$MethodCount = 0; [int]$MaxMethodLines = 0
    [double]$AvgMethodLines = 0.0; [int]$BranchCount = 0; [int]$MaxNestDepth = 0; [int]$GVRefCount = 0
}

# ============================================================
# パーサー（テキスト版と同一ロジック）
# ============================================================

class VBNetParser {
    [string]$ProjectPath
    [System.Collections.Generic.List[string]]$Files = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[FileMetrics]]$FileMetrics = [System.Collections.Generic.List[FileMetrics]]::new()
    [System.Collections.Generic.List[MethodInfo]]$Methods = [System.Collections.Generic.List[MethodInfo]]::new()
    [System.Collections.Generic.Dictionary[string, GlobalVarInfo]]$GlobalVars = [System.Collections.Generic.Dictionary[string, GlobalVarInfo]]::new()
    [System.Collections.Generic.Dictionary[string, FormInfo]]$Forms = [System.Collections.Generic.Dictionary[string, FormInfo]]::new()
    [System.Collections.Generic.List[BranchInfo]]$Branches = [System.Collections.Generic.List[BranchInfo]]::new()
    [hashtable]$AllLines = @{}

    static [string[]]$DBKeywords = @(
        'sqlceconnection','sqlcecommand','sqlcedatareader','sqlcedataadapter',
        'sqlconnection','sqlcommand','sqldatareader','sqldataadapter',
        'executenonquery','executereader','executescalar',
        'oledbconnection','oledbcommand',
        'select ','insert ','update ','delete ','create table',
        '.open()','.fill(','dataset','datatable')
    static [string[]]$NetKeywords = @(
        'httpwebrequest','httpwebresponse','webclient','webrequest',
        'tcpclient','udpclient','socket','serialport','system.net',
        '.getresponse(','.downloadstring(','.uploadstring(')

    VBNetParser([string]$path) { $this.ProjectPath = $path }

    [void]DiscoverFiles() {
        Get-ChildItem -Path $this.ProjectPath -Filter '*.vb' -Recurse -File | ForEach-Object { $this.Files.Add($_.FullName) }
        $this.Files.Sort()
    }

    [string[]]ReadFile([string]$fp) {
        foreach ($enc in @('utf-8','shift_jis','Default')) {
            try { return [System.IO.File]::ReadAllLines($fp, [System.Text.Encoding]::GetEncoding($enc)) } catch { continue }
        }
        try { return [System.IO.File]::ReadAllLines($fp) } catch { return @() }
    }

    [string]ClassifyFile([string]$fp, [string]$content) {
        $fn = [IO.Path]::GetFileName($fp).ToLower()
        if ($fn.EndsWith('.designer.vb')) { return 'Designer' }
        $cl = $content.ToLower()
        if ($cl.Contains('inherits system.windows.forms.form') -or $cl.Contains('inherits form')) { return 'Form' }
        if ($cl.Contains('inherits system.windows.forms.usercontrol') -or $cl.Contains('inherits usercontrol')) { return 'UserControl' }
        if ($cl -match '(?m)^\s*module\s+') { return 'Module' }
        return 'Class'
    }

    [void]CollectGlobalVars() {
        $reGV = [regex]::new('^\s*(Public|Friend)\s+(Shared\s+)?(Dim\s+|Const\s+)?(\w+)\s+As\s+(\w[\w\(\)]*)', 'IgnoreCase')
        $reC = [regex]::new('^\s*(Public|Private|Friend|Protected)?\s*(Partial\s+)?(Class|Module|Structure)\s+(\w+)', 'IgnoreCase')
        $reEC = [regex]::new('^\s*End\s+(Class|Module|Structure)', 'IgnoreCase')
        foreach ($fp in $this.Files) {
            $lines = $this.AllLines[$fp]; if (-not $lines) { continue }
            $fname = [IO.Path]::GetFileName($fp); $inC = $false; $curC = ''
            foreach ($line in $lines) {
                $s = $line.Trim()
                $mc = $reC.Match($s); if ($mc.Success) { $inC = $true; $curC = $mc.Groups[4].Value; continue }
                if ($reEC.IsMatch($s)) { $inC = $false }
                $mg = $reGV.Match($s)
                if ($mg.Success -and $inC) {
                    $vn = $mg.Groups[4].Value; $key = $vn.ToLower()
                    if (-not $this.GlobalVars.ContainsKey($key)) {
                        $gv = [GlobalVarInfo]::new(); $gv.Name = $vn; $gv.VarType = $mg.Groups[5].Value
                        $gv.DeclaredIn = "$fname ($curC)"; $gv.Modifier = $mg.Groups[1].Value
                        $im = [regex]::Match($s, '=\s*(.+)$'); if ($im.Success) { $gv.InitialValue = $im.Groups[1].Value.Trim() }
                        $this.GlobalVars[$key] = $gv
                    }
                }
            }
        }
    }

    [void]Analyze() {
        foreach ($fp in $this.Files) { $this.AllLines[$fp] = $this.ReadFile($fp) }
        $this.CollectGlobalVars()
        foreach ($fp in $this.Files) {
            $lines = $this.AllLines[$fp]; $content = $lines -join "`n"
            $fname = [IO.Path]::GetFileName($fp); $ftype = $this.ClassifyFile($fp, $content)
            $fm = [FileMetrics]::new(); $fm.FilePath = $fp; $fm.FileName = $fname; $fm.FileType = $ftype
            $this.CalcLineMetrics($lines, $fm)
            if ($ftype -ne 'Designer') {
                $this.ParseMethods($fp, $lines, $fm); $this.ScanGVRefs($fp, $lines, $fm)
                if ($ftype -in @('Form','UserControl')) { $this.ParseForm($fp, $lines, $content) }
            }
            $this.FileMetrics.Add($fm)
        }
    }

    [void]CalcLineMetrics([string[]]$lines, [FileMetrics]$fm) {
        $fm.TotalLines = $lines.Count
        foreach ($l in $lines) {
            $s = $l.Trim()
            if ([string]::IsNullOrEmpty($s)) { $fm.BlankLines++ }
            elseif ($s.StartsWith("'") -or $s.ToUpper().StartsWith('REM ')) { $fm.CommentLines++ }
            else { $fm.CodeLines++ }
        }
    }

    [void]ParseMethods([string]$fp, [string[]]$lines, [FileMetrics]$fm) {
        $fname = [IO.Path]::GetFileName($fp)
        $reC = [regex]::new('^\s*(Public|Private|Friend|Protected)?\s*(Partial\s+)?(Class|Module|Structure)\s+(\w+)', 'IgnoreCase')
        $reM = [regex]::new('^\s*(Public|Private|Protected|Friend)?\s*(Shared\s+)?(Overrides\s+)?(Sub|Function)\s+(\w+)\s*\(([^)]*)\)(?:\s+As\s+(\w+))?', 'IgnoreCase')
        $reEB = [regex]::new('^\s*End\s+(Sub|Function|Class|Module|Structure|Property|If|Select|While|For|Try|With|Using|SyncLock)', 'IgnoreCase')
        $reIf = [regex]::new('^\s*If\s+(.+?)\s+Then\s*$', 'IgnoreCase')
        $reEI = [regex]::new('^\s*ElseIf\s+(.+?)\s+Then', 'IgnoreCase')
        $reSC = [regex]::new('^\s*Select\s+Case\s+(.+)', 'IgnoreCase')
        $reCs = [regex]::new('^\s*Case\s+(.+)', 'IgnoreCase')
        $reFor = [regex]::new('^\s*For\s+', 'IgnoreCase')
        $reFE = [regex]::new('^\s*For\s+Each\s+', 'IgnoreCase')
        $reW = [regex]::new('^\s*(?:While|Do\s+While|Do\s+Until)\s+', 'IgnoreCase')
        $reT = [regex]::new('^\s*Try\s*$', 'IgnoreCase')
        $reCa = [regex]::new('^\s*Catch\b', 'IgnoreCase')
        $reCP = [regex]::new('(\w+)\.(Text|Visible|Enabled|BackColor|ForeColor|Caption|Name|DataSource|Columns)\s*=\s*(.+)', 'IgnoreCase')

        $curC = ''; [System.Collections.Generic.List[object]]$mStack = @(); $nest = 0

        for ($i = 0; $i -lt $lines.Count; $i++) {
            $code = ($lines[$i] -replace "'.*$", '').Trim()
            if ([string]::IsNullOrEmpty($code)) { continue }
            $ln = $i + 1

            $mc = $reC.Match($code); if ($mc.Success) { $curC = $mc.Groups[4].Value; $fm.ClassCount++; continue }
            $mm = $reM.Match($code)
            if ($mm.Success) {
                $mi = [MethodInfo]::new(); $mi.Name = $mm.Groups[5].Value; $mi.FilePath = $fname; $mi.ClassName = $curC
                $mi.AccessModifier = if ($mm.Groups[1].Value) { $mm.Groups[1].Value } else { 'Private' }
                $mi.ReturnType = if ($mm.Groups[7].Value) { $mm.Groups[7].Value } elseif ($mm.Groups[4].Value -ieq 'sub') { 'Void' } else { '?' }
                $mi.Params = ($mm.Groups[6].Value ?? '').Trim(); $mi.StartLine = $ln; $mi.EndLine = $ln
                $mStack.Add(@{ M = $mi; B = $nest }); $nest++; $fm.MethodCount++; continue
            }

            $curM = if ($mStack.Count -gt 0) { $mStack[-1].M } else { $null }

            $mIf = $reIf.Match($code)
            if ($mIf.Success) {
                $nest++
                if ($curM) {
                    $cond = $mIf.Groups[1].Value; $curM.BranchCount++; $curM.CyclomaticComplexity++; $fm.BranchCount++
                    $bi = [BranchInfo]::new(); $bi.FilePath = $fname; $bi.MethodName = $curM.Name; $bi.LineNumber = $ln
                    $bi.BranchType = 'If'; $bi.Condition = $cond.Substring(0,[Math]::Min($cond.Length,120)); $bi.NestDepth = $nest
                    $this.Branches.Add($bi)
                    $curM.CyclomaticComplexity += [regex]::Matches($cond, '\b(And|Or|AndAlso|OrElse)\b', 'IgnoreCase').Count
                }
            } elseif ($reEI.IsMatch($code)) {
                if ($curM) {
                    $cond = $reEI.Match($code).Groups[1].Value; $curM.BranchCount++; $curM.CyclomaticComplexity++; $fm.BranchCount++
                    $bi = [BranchInfo]::new(); $bi.FilePath = $fname; $bi.MethodName = $curM.Name; $bi.LineNumber = $ln
                    $bi.BranchType = 'ElseIf'; $bi.Condition = $cond.Substring(0,[Math]::Min($cond.Length,120)); $bi.NestDepth = $nest
                    $this.Branches.Add($bi)
                }
            } elseif ($reSC.IsMatch($code)) {
                $nest++
                if ($curM) {
                    $cond = $reSC.Match($code).Groups[1].Value; $curM.BranchCount++; $curM.CyclomaticComplexity++; $fm.BranchCount++
                    $bi = [BranchInfo]::new(); $bi.FilePath = $fname; $bi.MethodName = $curM.Name; $bi.LineNumber = $ln
                    $bi.BranchType = 'Select Case'; $bi.Condition = $cond.Substring(0,[Math]::Min($cond.Length,120)); $bi.NestDepth = $nest
                    $this.Branches.Add($bi)
                }
            } elseif ($reCs.IsMatch($code) -and -not $code.Trim().ToLower().StartsWith('select')) {
                if ($curM -and -not $code.Trim().ToLower().StartsWith('case else')) { $curM.CyclomaticComplexity++ }
            } elseif ($reFor.IsMatch($code) -or $reFE.IsMatch($code)) {
                $nest++; if ($curM) { $curM.CyclomaticComplexity++ }
            } elseif ($reW.IsMatch($code)) {
                $nest++; if ($curM) { $curM.CyclomaticComplexity++ }
            } elseif ($reT.IsMatch($code)) { $nest++ }
            elseif ($reCa.IsMatch($code)) { if ($curM) { $curM.CyclomaticComplexity++ } }

            if ($mStack.Count -gt 0) {
                $rel = $nest - $mStack[-1].B - 1
                if ($rel -gt $curM.MaxNestDepth) { $curM.MaxNestDepth = $rel }
            }

            if ($curM) {
                $cp = $reCP.Match($code)
                if ($cp.Success) {
                    for ($bi_i = $this.Branches.Count - 1; $bi_i -ge 0; $bi_i--) {
                        $b = $this.Branches[$bi_i]
                        if ($b.FilePath -eq $fname -and $b.MethodName -eq $curM.Name -and [string]::IsNullOrEmpty($b.Target)) {
                            $b.Target = "$($cp.Groups[1].Value).$($cp.Groups[2].Value)"; break
                        }
                    }
                }
                $cl = $code.ToLower()
                foreach ($kw in [VBNetParser]::DBKeywords) { if ($cl.Contains($kw)) { $curM.HasDBAccess = $true; break } }
                foreach ($kw in [VBNetParser]::NetKeywords) { if ($cl.Contains($kw)) { $curM.HasNetAccess = $true; break } }
            }

            $em = $reEB.Match($code)
            if ($em.Success) {
                $bt = $em.Groups[1].Value.ToLower()
                if ($bt -in @('sub','function') -and $mStack.Count -gt 0) {
                    $entry = $mStack[-1]; $mi = $entry.M; $mStack.RemoveAt($mStack.Count - 1)
                    $mi.EndLine = $ln; $mi.LineCount = $mi.EndLine - $mi.StartLine + 1
                    $this.ScanMethodBody($fp, $lines, $mi); $this.Methods.Add($mi)
                }
                $nest = [Math]::Max(0, $nest - 1)
            }
        }
        $ml = $this.Methods | Where-Object { $_.FilePath -eq $fname } | ForEach-Object { $_.LineCount }
        if ($ml) { $fm.MaxMethodLines = ($ml | Measure-Object -Maximum).Maximum; $fm.AvgMethodLines = ($ml | Measure-Object -Average).Average }
        $ns = $this.Methods | Where-Object { $_.FilePath -eq $fname } | ForEach-Object { $_.MaxNestDepth }
        if ($ns) { $fm.MaxNestDepth = ($ns | Measure-Object -Maximum).Maximum }
    }

    [void]ScanMethodBody([string]$fp, [string[]]$lines, [MethodInfo]$mi) {
        $skip = [System.Collections.Generic.HashSet[string]]::new(
            [string[]]@('if','while','for','select','case','catch','ctype','cstr','cint','cdbl','cbool','directcast','trycast','typeof','gettype','nothing','not','and','or'),
            [System.StringComparer]::OrdinalIgnoreCase)
        $si = [Math]::Max(0, $mi.StartLine - 1); $ei = [Math]::Min($mi.EndLine, $lines.Count)
        for ($li = $si; $li -lt $ei; $li++) {
            $code = ($lines[$li] -replace "'.*$", '').Trim(); if ([string]::IsNullOrEmpty($code)) { continue }
            $loc = "$($mi.FilePath):$($li+1)"
            foreach ($key in $this.GlobalVars.Keys) {
                $gv = $this.GlobalVars[$key]
                if ($code -match "\b$([regex]::Escape($gv.Name))\b") {
                    if ($code -match "\b$([regex]::Escape($gv.Name))\s*=") {
                        if (-not $mi.GVWrite.Contains($gv.Name)) { $mi.GVWrite.Add($gv.Name) }
                        if (-not $gv.SetLocations.Contains($loc)) { $gv.SetLocations.Add($loc) }
                    } else {
                        if (-not $mi.GVRead.Contains($gv.Name)) { $mi.GVRead.Add($gv.Name) }
                        if (-not $gv.ReadLocations.Contains($loc)) { $gv.ReadLocations.Add($loc) }
                    }
                    [void]$gv.RefFunctions.Add("$($mi.ClassName).$($mi.Name)")
                    [void]$gv.RefFiles.Add($mi.FilePath)
                }
            }
            foreach ($cm in [regex]::Matches($code, '(\w+\.\w+|\w+)\s*\(')) {
                $call = $cm.Groups[1].Value
                if (-not $skip.Contains($call) -and -not $mi.CallsTo.Contains($call)) { $mi.CallsTo.Add($call) }
            }
        }
    }

    [void]ScanGVRefs([string]$fp, [string[]]$lines, [FileMetrics]$fm) {
        $c = 0
        foreach ($l in $lines) { $code = ($l -replace "'.*$",'').Trim(); foreach ($gv in $this.GlobalVars.Values) { if ($code -match "\b$([regex]::Escape($gv.Name))\b") { $c++ } } }
        $fm.GVRefCount = $c
    }

    [void]ParseForm([string]$fp, [string[]]$lines, [string]$content) {
        $fname = [IO.Path]::GetFileName($fp); $formName = [IO.Path]::GetFileNameWithoutExtension($fname)
        $fi = [FormInfo]::new(); $fi.Name = $formName; $fi.FilePath = $fname
        $reC1 = [regex]::new('(?:Me\.)?(\w+)\s*=\s*New\s+System\.Windows\.Forms\.(\w+)','IgnoreCase')
        $reC2 = [regex]::new('Friend\s+WithEvents\s+(\w+)\s+As\s+(?:System\.Windows\.Forms\.)?(\w+)','IgnoreCase')
        $reH = [regex]::new('(?:Private|Public|Protected)\s+Sub\s+(\w+_\w+)\s*\(','IgnoreCase')
        $reS = [regex]::new('(\w+)\.(Show|ShowDialog)\b','IgnoreCase')
        $reP = [regex]::new('(\w+)\.(Text|Visible|Enabled|BackColor|ForeColor|Caption|Name|DataSource|Columns)\s*=\s*(.+)','IgnoreCase')
        $all = [System.Collections.Generic.List[string]]::new($lines)
        $dp = $fp -replace '\.vb$','.Designer.vb'; if (Test-Path $dp) { $all.AddRange($this.ReadFile($dp)) }
        $ctrls = [System.Collections.Generic.HashSet[string]]::new()
        foreach ($l in $all) { foreach ($re in @($reC1,$reC2)) { $m = $re.Match($l); if ($m.Success) { [void]$ctrls.Add("$($m.Groups[1].Value) ($($m.Groups[2].Value))") } } }
        $fi.Controls.AddRange([string[]]($ctrls | Sort-Object))
        foreach ($l in $lines) { $m = $reH.Match($l.Trim()); if ($m.Success) { $fi.EventHandlers.Add($m.Groups[1].Value) }; $m = $reS.Match($l); if ($m.Success -and -not $fi.NavigatesTo.Contains($m.Groups[1].Value)) { $fi.NavigatesTo.Add($m.Groups[1].Value) } }
        for ($li = 0; $li -lt $lines.Count; $li++) {
            $code = ($lines[$li] -replace "'.*$",'').Trim(); $m = $reP.Match($code)
            if ($m.Success) { $val = $m.Groups[3].Value.Trim(); $fi.DynamicControls.Add(@{ Control = $m.Groups[1].Value; Property = $m.Groups[2].Value; Value = $val.Substring(0,[Math]::Min($val.Length,60)); Line = $li+1 }) }
        }
        $this.Forms[$formName] = $fi
    }
}

# ============================================================
# Excel COM レポート生成
# ============================================================

function New-ExcelReport {
    param([VBNetParser]$Parser, [string]$OutputPath)

    $excel = $null; $wb = $null
    try {
        Write-Host '  Excel起動中...' -ForegroundColor Gray
        $excel = New-Object -ComObject Excel.Application
        $excel.Visible = $false
        $excel.DisplayAlerts = $false
        $excel.ScreenUpdating = $false
        $wb = $excel.Workbooks.Add()

        # 不要な既定シート削除（1枚残す）
        while ($wb.Sheets.Count -gt 1) { $wb.Sheets.Item($wb.Sheets.Count).Delete() }

        # --- 色定数 ---
        $cNavy    = 0x64381F  # BGR: 1F3864
        $cDkBlue  = 0xB6752E  # 2E75B6
        $cMidBlue = 0xC47244  # 4472C4
        $cLtBlue  = 0xF0E4D6  # D6E4F0
        $cWhite   = 0xFFFFFF
        $cLtGray  = 0xF2F2F2
        $cYellow  = 0xCCF2FF  # FFF2CC
        $cRed     = 0xE4E4FC  # FCE4EC
        $cGreen   = 0xDAEFE2  # E2EFDA
        $cOrange  = 0xD6E5FB  # FBE5D6
        $cDkRed   = 0x0000C0  # C00000

        # --- ヘルパー関数 ---
        function Set-HeaderRow($ws, [int]$row, [string[]]$headers, $bgColor) {
            if (-not $bgColor) { $bgColor = $cNavy }
            for ($c = 0; $c -lt $headers.Count; $c++) {
                $cell = $ws.Cells.Item($row, $c + 1)
                $cell.Value2 = $headers[$c]
                $cell.Font.Bold = $true; $cell.Font.Size = 10; $cell.Font.Name = 'Arial'
                $cell.Font.Color = $cWhite
                $cell.Interior.Color = $bgColor
                $cell.HorizontalAlignment = -4108  # xlCenter
                $cell.VerticalAlignment = -4108
                $cell.WrapText = $true
                $cell.Borders.LineStyle = 1; $cell.Borders.Color = 0xB0B0B0
            }
        }

        function Set-DataCell($ws, [int]$row, [int]$col, $value, [bool]$alt) {
            $cell = $ws.Cells.Item($row, $col)
            if ($null -ne $value) { $cell.Value2 = $value } else { $cell.Value2 = '' }
            $cell.Font.Name = 'Arial'; $cell.Font.Size = 10
            if ($alt) { $cell.Interior.Color = $cLtGray } else { $cell.Interior.Color = $cWhite }
            $cell.Borders.LineStyle = 1; $cell.Borders.Color = 0xB0B0B0
            $cell.VerticalAlignment = -4108
            $cell.WrapText = $true
            return $cell
        }

        function Set-Title($ws, [string]$title, [string]$subtitle, [int]$cols) {
            $ws.Range($ws.Cells.Item(1,1), $ws.Cells.Item(1,$cols)).Merge()
            $ws.Cells.Item(1,1).Value2 = $title
            $ws.Cells.Item(1,1).Font.Bold = $true; $ws.Cells.Item(1,1).Font.Size = 14
            $ws.Cells.Item(1,1).Font.Name = 'Arial'; $ws.Cells.Item(1,1).Font.Color = $cNavy
            $ws.Rows.Item(1).RowHeight = 28
            $ws.Range($ws.Cells.Item(2,1), $ws.Cells.Item(2,$cols)).Merge()
            $ws.Cells.Item(2,1).Value2 = $subtitle
            $ws.Cells.Item(2,1).Font.Italic = $true; $ws.Cells.Item(2,1).Font.Size = 9
            $ws.Cells.Item(2,1).Font.Color = 0x666666
            $ws.Range($ws.Cells.Item(3,1), $ws.Cells.Item(3,$cols)).Merge()
            $ws.Cells.Item(3,1).Value2 = "解析日時: $(Get-Date -Format 'yyyy-MM-dd HH:mm')  |  対象: $($Parser.ProjectPath)"

                $ws8.Cells.Item($hr, $j+2).Borders.LineStyle = 1
                $ws8.Cells.Item($hr, $j+2).Orientation = 45
            }
            for ($i = 0; $i -lt $n; $i++) {
                $ri = $hr + 1 + $i
                $ws8.Cells.Item($ri, 1).Value2 = $files[$i]; $ws8.Cells.Item($ri, 1).Font.Bold = $true
                $ws8.Cells.Item($ri, 1).Interior.Color = $cLtBlue; $ws8.Cells.Item($ri, 1).Borders.LineStyle = 1
                for ($j = 0; $j -lt $n; $j++) {
                    $v = $mat[$i][$j]; $cell = $ws8.Cells.Item($ri, $j+2)
                    if ($v -gt 0) { $cell.Value2 = $v } else { $cell.Value2 = '' }
                    $cell.HorizontalAlignment = -4108; $cell.Borders.LineStyle = 1
                    if ($i -eq $j -and $v -gt 0) { $cell.Interior.Color = $cLtBlue }
                    elseif ($v -ge 10) { $cell.Interior.Color = $cRed }
                    elseif ($v -ge 5) { $cell.Interior.Color = $cOrange }
                    elseif ($v -gt 0) { $cell.Interior.Color = $cGreen }
                }
            }
            $ws8.Columns.Item(1).ColumnWidth = 22
            for ($j = 2; $j -le $n + 1; $j++) { $ws8.Columns.Item($j).ColumnWidth = 14 }
        }

        # ================================================================
        # 保存
        # ================================================================
        Write-Host '  保存中...' -ForegroundColor Gray
        $ws1.Select() | Out-Null
        $excel.ScreenUpdating = $true
        $wb.SaveAs($OutputPath, 51)  # xlOpenXMLWorkbook
        Write-Host "  保存完了: $OutputPath" -ForegroundColor Green
    }
    catch {
        Write-Error "Excel COM エラー: $_"
    }
    finally {
        if ($wb) {
            try { $wb.Close($false) } catch {}
            [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($wb)
        }
        if ($excel) {
            try { $excel.Quit() } catch {}
            [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($excel)
        }
        [GC]::Collect(); [GC]::WaitForPendingFinalizers()
    }
}
