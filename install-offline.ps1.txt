# Next.js オフラインインストールスクリプト
# 使い方: powershell -ExecutionPolicy Bypass -File install-offline.ps1

param(
    [string]$packagesDir = "npm-packages"
)

Write-Host "========================================" -ForegroundColor Cyan
Write-Host "  Next.js Offline Installer" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host ""

# 必要なファイルの確認
$requiredFiles = @("package.json", "package-lock.json")
$missingFiles = @()

foreach ($file in $requiredFiles) {
    if (-not (Test-Path $file)) {
        $missingFiles += $file
    }
}

if ($missingFiles.Count -gt 0) {
    Write-Host "Error: Required files are missing:" -ForegroundColor Red
    foreach ($file in $missingFiles) {
        Write-Host "  - $file" -ForegroundColor Red
    }
    exit 1
}

# パッケージディレクトリの確認
if (-not (Test-Path $packagesDir)) {
    Write-Host "Error: Package directory not found: $packagesDir" -ForegroundColor Red
    Write-Host "Please make sure you've copied the npm-packages folder." -ForegroundColor Yellow
    exit 1
}

$tgzFiles = Get-ChildItem -Path $packagesDir -Filter "*.tgz"
if ($tgzFiles.Count -eq 0) {
    Write-Host "Error: No .tgz files found in $packagesDir" -ForegroundColor Red
    exit 1
}

Write-Host "Found $($tgzFiles.Count) package files in $packagesDir" -ForegroundColor Green
Write-Host ""

# npmがインストールされているか確認
try {
    $npmVersion = npm --version
    Write-Host "npm version: $npmVersion" -ForegroundColor Green
} catch {
    Write-Host "Error: npm is not installed or not in PATH" -ForegroundColor Red
    Write-Host "Please install Node.js first: https://nodejs.org/" -ForegroundColor Yellow
    exit 1
}

Write-Host ""
Write-Host "Starting offline installation..." -ForegroundColor Cyan
Write-Host "This may take several minutes..." -ForegroundColor Yellow
Write-Host ""

# 方法1: npmのオフラインキャッシュを使用
$cacheDir = ".npm-offline-cache"

# キャッシュディレクトリを作成
if (Test-Path $cacheDir) {
    Write-Host "Cleaning old cache..." -ForegroundColor Yellow
    Remove-Item -Path $cacheDir -Recurse -Force
}
New-Item -ItemType Directory -Path $cacheDir | Out-Null

Write-Host "Preparing npm cache..." -ForegroundColor Cyan

# すべてのtgzファイルをnpmキャッシュに追加
foreach ($tgz in $tgzFiles) {
    try {
        # npmキャッシュにパッケージを追加
        npm cache add $tgz.FullName --cache $cacheDir 2>&1 | Out-Null
    } catch {
        Write-Host "Warning: Failed to cache $($tgz.Name)" -ForegroundColor Yellow
    }
}

Write-Host "Cache prepared. Installing packages..." -ForegroundColor Cyan
Write-Host ""

# node_modulesが既に存在する場合は削除
if (Test-Path "node_modules") {
    Write-Host "Removing existing node_modules..." -ForegroundColor Yellow
    Remove-Item -Path "node_modules" -Recurse -Force
}

# オフラインモードでインストール
try {
    # --prefer-offline: キャッシュを優先
    # --no-audit: 監査スキップ（オフラインなので）
    # --cache: カスタムキャッシュディレクトリを使用
    npm install --prefer-offline --no-audit --cache $cacheDir
    
    if ($LASTEXITCODE -eq 0) {
        Write-Host ""
        Write-Host "========================================" -ForegroundColor Cyan
        Write-Host "  Installation Successful!" -ForegroundColor Green
        Write-Host "========================================" -ForegroundColor Cyan
        Write-Host ""
        Write-Host "Your Next.js project is ready!" -ForegroundColor Green
        Write-Host ""
        Write-Host "Next steps:" -ForegroundColor White
        Write-Host "  1. Run 'npm run dev' to start development server" -ForegroundColor White
        Write-Host "  2. Open http://localhost:3000 in your browser" -ForegroundColor White
        Write-Host ""
        
        # インストールされたパッケージ数を確認
        if (Test-Path "node_modules") {
            $installedCount = (Get-ChildItem -Path "node_modules" -Directory).Count
            Write-Host "Installed packages: $installedCount" -ForegroundColor Cyan
        }
    } else {
        Write-Host ""
        Write-Host "Installation failed with exit code: $LASTEXITCODE" -ForegroundColor Red
        Write-Host ""
        Write-Host "Troubleshooting:" -ForegroundColor Yellow
        Write-Host "  1. Make sure all .tgz files are present in $packagesDir" -ForegroundColor White
        Write-Host "  2. Check that package-lock.json matches the downloaded packages" -ForegroundColor White
        Write-Host "  3. Try running the script again" -ForegroundColor White
    }
} catch {
    Write-Host ""
    Write-Host "Error during installation:" -ForegroundColor Red
    Write-Host $_.Exception.Message -ForegroundColor Red
}
