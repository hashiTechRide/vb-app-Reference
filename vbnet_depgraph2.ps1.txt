#Requires -Version 7.0
<#
.SYNOPSIS
    VB.NET 2008 CE 依存関係グラフ ビジュアライザー
.DESCRIPTION
    指定ファイル/メソッドを起点に、依存関係をインタラクティブHTMLで可視化。
    ファイル間呼出・GV読書き・クラス階層・画面遷移を1つのグラフに統合表示。
    出力: HTML (D3.js force graph)  依存: なし (PwSh7標準のみ)
.PARAMETER Path
    プロジェクトフォルダ
.PARAMETER Focus
    起点ファイル名またはメソッド名（省略時は全体表示）
    例: -Focus "frmScan.vb"  /  -Focus "DoNyuko"  /  -Focus "frmScan.vb","modBL.vb"
.PARAMETER Depth
    起点からの探索深度（既定: 3）
.PARAMETER OutFile
    出力HTMLパス（省略時は自動命名）
.EXAMPLE
    .\vbnet_depgraph.ps1 -Path "C:\Projects\App"
    .\vbnet_depgraph.ps1 -Path "C:\Projects\App" -Focus "frmScan.vb" -Depth 2
    .\vbnet_depgraph.ps1 -Path "C:\Projects\App" -Focus "DoNyuko","g_FuncMode"
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory,Position=0)][ValidateScript({Test-Path $_ -PathType Container})][string]$Path,
    [string[]]$Focus,
    [int]$Depth = 3,
    [string]$OutFile
)
$ErrorActionPreference = 'Stop'

function Read-VBFile([string]$fp) {
    foreach($e in @('utf-8','shift_jis','Default')){try{return [IO.File]::ReadAllLines($fp,[Text.Encoding]::GetEncoding($e))}catch{continue}}
    try{return [IO.File]::ReadAllLines($fp)}catch{return @()}
}

# ============================================================
# 解析
# ============================================================
function Invoke-DependencyParse([string]$ProjectPath) {
    $allFiles = @(Get-ChildItem -Path $ProjectPath -Filter '*.vb' -Recurse -File | Sort-Object FullName)
    $nodes = [System.Collections.Generic.Dictionary[string,hashtable]]::new()
    $edges = [System.Collections.Generic.List[hashtable]]::new()

    # Regex
    $reClass  = [regex]::new('^\s*(Public|Private|Friend|Protected)?\s*(Partial\s+)?(Class|Module|Structure)\s+(\w+)','IgnoreCase')
    $reEndCls = [regex]::new('^\s*End\s+(Class|Module|Structure)','IgnoreCase')
    $reMethod = [regex]::new('^\s*(Public|Private|Protected|Friend)?\s*(?:Shared\s+)?(?:Overrides\s+)?(?:Sub|Function)\s+(\w+)\s*\(([^)]*)\)','IgnoreCase')
    $reEndM   = [regex]::new('^\s*End\s+(Sub|Function)','IgnoreCase')
    $reGVDecl = [regex]::new('^\s*(Public|Friend)\s+(?:Shared\s+)?(?:Dim\s+|Const\s+)?(\w+)\s+As\s+(\w+)','IgnoreCase')
    $reShow   = [regex]::new('(\w+)\.(Show|ShowDialog)\b','IgnoreCase')
    $reCall   = [regex]::new('(\w+)\.(\w+)\s*\(','IgnoreCase')
    $reCtrl   = [regex]::new('Friend\s+WithEvents\s+(\w+)\s+As\s+(?:System\.Windows\.Forms\.)?(\w+)','IgnoreCase')
    $reInherits=[regex]::new('Inherits\s+(System\.Windows\.Forms\.)?(Form|UserControl)\b','IgnoreCase')

    $skipCalls = [System.Collections.Generic.HashSet[string]]::new([string[]]@(
        'Me','MyBase','System','Math','Convert','String','Integer','Boolean','Double',
        'CType','CStr','CInt','CDbl','CBool','DirectCast','TryCast','MessageBox',
        'DateTime','Nothing','Array','ArrayList','Hashtable','Console','Debug',
        'Environment','IO','Text','Threading','Net','Data'),
        [StringComparer]::OrdinalIgnoreCase)

    # Global vars
    $globalVars = [System.Collections.Generic.Dictionary[string,string]]::new([StringComparer]::OrdinalIgnoreCase)

    # ---- Pass 1: ノード構築 (ファイル, クラス, メソッド, GV) ----
    Write-Host '  Pass 1: ノード構築...' -ForegroundColor Gray
    $classToFile = [ordered]@{}
    $methodOwner = [ordered]@{}  # methodName -> className

    foreach ($f in $allFiles) {
        $fn = $f.Name
        $lines = Read-VBFile $f.FullName
        $isDesigner = $fn -like '*.Designer.vb'
        $content = $lines -join "`n"
        $isForm = $content -match 'Inherits\s+(System\.Windows\.Forms\.)?(Form|UserControl)' -or $content -match 'Inherits\s+Form\b'

        # File node
        $fType = if ($isDesigner){'designer'} elseif($isForm){'form'} elseif($content -match '(?m)^\s*Module\s+'){'module'} else{'class'}
        if (-not $isDesigner) {
            $fid = "file:$fn"
            $nodes[$fid] = @{id=$fid; label=$fn; type=$fType; group='file'; level=0}
        }

        if ($isDesigner) {
            # Designer: コントロールだけ拾う
            foreach ($line in $lines) {
                $mc = $reCtrl.Match($line)
                if ($mc.Success) {
                    $parentFile = $fn -replace '\.Designer\.vb$','.vb'
                    # Ctrl情報は親ファイルに紐づけるだけ（ノード追加は重くなるので省略）
                }
            }
            continue
        }

        $curClass = ''; $curMethod = $null; $methodStart = 0
        foreach ($line in $lines) {
            $code = ($line -replace "'.*$",'').Trim()
            if ([string]::IsNullOrEmpty($code)) { continue }

            $mc = $reClass.Match($code)
            if ($mc.Success) {
                $curClass = $mc.Groups[4].Value
                $classToFile[$curClass] = $fn
                $cid = "class:$curClass"
                $cType = if($code -match '^\s*(?:Public\s+)?Module\s+'){'module'} else{'class'}
                $nodes[$cid] = @{id=$cid; label=$curClass; type=$cType; group='class'; level=1; parent="file:$fn"}
                # File -> Class edge
                $edges.Add(@{source="file:$fn"; target=$cid; type='contains'; label='contains'})
                continue
            }

            # GV
            $mg = $reGVDecl.Match($code)
            if ($mg.Success -and $curClass) {
                $gvName = $mg.Groups[2].Value
                $gvType = $mg.Groups[3].Value
                $globalVars[$gvName] = $curClass
                $gid = "gv:$gvName"
                if (-not $nodes.ContainsKey($gid)) {
                    $nodes[$gid] = @{id=$gid; label="GV:$gvName"; type='globalvar'; group='gv'; level=2; varType=$gvType; parent="class:$curClass"}
                }
            }

            # Method
            $mm = $reMethod.Match($code)
            if ($mm.Success -and $curClass) {
                $mName = $mm.Groups[2].Value
                $curMethod = $mName; $methodStart = 0
                $mid = "method:$curClass.$mName"
                $acc = if($mm.Groups[1].Value){"$($mm.Groups[1].Value)"}else{'Private'}
                $methodOwner[$mName] = $curClass
                if (-not $nodes.ContainsKey($mid)) {
                    $nodes[$mid] = @{id=$mid; label="$curClass.$mName"; type='method'; group='method'; level=2; access=$acc; parent="class:$curClass"}
                    $edges.Add(@{source="class:$curClass"; target=$mid; type='contains'; label='has'})
                }
            }

            $me = $reEndM.Match($code)
            if ($me.Success) { $curMethod = $null }
        }
    }

    # ---- Pass 2: エッジ構築 (呼出, GV参照, 画面遷移) ----
    Write-Host '  Pass 2: エッジ構築...' -ForegroundColor Gray

    foreach ($f in $allFiles) {
        $fn = $f.Name
        if ($fn -like '*.Designer.vb') { continue }
        $lines = Read-VBFile $f.FullName
        $curClass = ''; $curMethod = ''

        foreach ($line in $lines) {
            $code = ($line -replace "'.*$",'').Trim()
            if ([string]::IsNullOrEmpty($code)) { continue }

            $mc = $reClass.Match($code)
            if ($mc.Success) { $curClass = $mc.Groups[4].Value; continue }
            if ($reEndCls.IsMatch($code)) { $curClass = ''; continue }

            $mm = $reMethod.Match($code)
            if ($mm.Success) { $curMethod = $mm.Groups[2].Value; continue }
            if ($reEndM.IsMatch($code)) { $curMethod = ''; continue }

            if (-not $curClass -or -not $curMethod) { continue }
            $srcMethod = "method:$curClass.$curMethod"

            # メソッド呼出: ClassName.MethodName(
            foreach ($m in $reCall.Matches($code)) {
                $callClass = $m.Groups[1].Value
                $callMethod = $m.Groups[2].Value
                if ($skipCalls.Contains($callClass)) { continue }

                # 解決: callClass がクラス名かどうか
                $resolvedClass = $null
                if ($classToFile.Contains($callClass)) { $resolvedClass = $callClass }
                elseif ($callClass -ieq 'Me') { $resolvedClass = $curClass }

                if ($resolvedClass) {
                    $tgtMethod = "method:$resolvedClass.$callMethod"
                    if ($nodes.ContainsKey($tgtMethod) -and $srcMethod -ne $tgtMethod) {
                        $edges.Add(@{source=$srcMethod; target=$tgtMethod; type='calls'; label='calls'})
                        # File間エッジも追加
                        $srcFile = "file:$fn"
                        $tgtFile = "file:$($classToFile[$resolvedClass])"
                        if ($srcFile -ne $tgtFile) {
                            $edges.Add(@{source=$srcFile; target=$tgtFile; type='file_calls'; label='calls'})
                        }
                    }
                }
            }

            # GV参照
            foreach ($gvName in $globalVars.Keys) {
                if ($code -match "\b$([regex]::Escape($gvName))\b") {
                    $gid = "gv:$gvName"
                    if ($code -match "\b$([regex]::Escape($gvName))\s*=(?!=)") {
                        $edges.Add(@{source=$srcMethod; target=$gid; type='gv_write'; label='writes'})
                    } else {
                        $edges.Add(@{source=$srcMethod; target=$gid; type='gv_read'; label='reads'})
                    }
                }
            }

            # 画面遷移: xxx.Show() / xxx.ShowDialog()
            foreach ($ms in $reShow.Matches($code)) {
                $tgtForm = $ms.Groups[1].Value
                # フォームの変数名からクラス名を推測 (New frmXxx パターン)
                # 直前行で Dim xxx As New frmYYY を探す...は複雑なので、変数名=クラス名と仮定
                $tgtFormFile = "$tgtForm.vb"
                $srcFormFile = "file:$fn"
                $tgtFormNode = "file:$tgtFormFile"
                if ($nodes.ContainsKey($tgtFormNode)) {
                    $edges.Add(@{source=$srcFormFile; target=$tgtFormNode; type='navigate'; label='navigate'})
                }
            }
        }
    }

    # 重複エッジ除去
    $edgeSet = [System.Collections.Generic.HashSet[string]]::new()
    $uniqueEdges = [System.Collections.Generic.List[hashtable]]::new()
    foreach ($e in $edges) {
        $key = "$($e.source)|$($e.target)|$($e.type)"
        if ($edgeSet.Add($key)) { $uniqueEdges.Add($e) }
    }

    Write-Host "  ノード: $($nodes.Count) / エッジ: $($uniqueEdges.Count)" -ForegroundColor Cyan
    return @{ Nodes = $nodes; Edges = $uniqueEdges; ClassToFile = $classToFile; MethodOwner = $methodOwner }
}

# ============================================================
# フォーカスフィルタリング
# ============================================================
function Get-FocusedSubgraph([hashtable]$Graph, [string[]]$FocusTargets, [int]$MaxDepth) {
    if (-not $FocusTargets -or $FocusTargets.Count -eq 0) { return $Graph }

    Write-Host "  フォーカスフィルタ: $($FocusTargets -join ', ') (深度$MaxDepth)" -ForegroundColor Cyan
    $included = [System.Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase)

    # フォーカス対象のノードIDを特定
    $seeds = [System.Collections.Generic.List[string]]::new()
    foreach ($ft in $FocusTargets) {
        foreach ($nid in $Graph.Nodes.Keys) {
            $node = $Graph.Nodes[$nid]
            if ($node.label -ieq $ft -or $nid -ieq "file:$ft" -or $nid -ieq "method:$ft" -or $nid -ieq "gv:$ft" -or $nid -ieq "class:$ft") {
                $seeds.Add($nid)
            }
            # 部分一致: "DoNyuko" -> "method:modBL.DoNyuko"
            if ($node.label -like "*$ft*" -or $node.label -like "*.$ft") {
                if (-not $seeds.Contains($nid)) { $seeds.Add($nid) }
            }
        }
    }

    if ($seeds.Count -eq 0) {
        Write-Host "  警告: フォーカス対象が見つかりません。全体表示します。" -ForegroundColor Yellow
        return $Graph
    }

    # BFS で深度制限付き探索
    $queue = [System.Collections.Generic.Queue[object]]::new()
    foreach ($s in $seeds) { $queue.Enqueue(@{Id=$s;Depth=0}); [void]$included.Add($s) }

    while ($queue.Count -gt 0) {
        $item = $queue.Dequeue()
        $curId = $item.Id; $curDepth = $item.Depth
        if ($curDepth -ge $MaxDepth) { continue }

        foreach ($e in $Graph.Edges) {
            $neighbor = $null
            if ($e.source -eq $curId) { $neighbor = $e.target }
            elseif ($e.target -eq $curId) { $neighbor = $e.source }
            if ($neighbor -and $included.Add($neighbor)) {
                $queue.Enqueue(@{Id=$neighbor;Depth=$curDepth+1})
            }
        }

        # 親ノード（class->file, method->class）も含める
        $node = $Graph.Nodes[$curId]
        if ($node -and $node.parent -and $included.Add($node.parent)) {
            $queue.Enqueue(@{Id=$node.parent;Depth=$curDepth})
        }
    }

    # フィルタ
    $fNodes = [System.Collections.Generic.Dictionary[string,hashtable]]::new()
    foreach ($nid in $included) {
        if ($Graph.Nodes.ContainsKey($nid)) { $fNodes[$nid] = $Graph.Nodes[$nid] }
    }
    $fEdges = [System.Collections.Generic.List[hashtable]]::new()
    foreach ($e in $Graph.Edges) {
        if ($included.Contains($e.source) -and $included.Contains($e.target)) { $fEdges.Add($e) }
    }

    Write-Host "  フィルタ後: ノード $($fNodes.Count) / エッジ $($fEdges.Count)" -ForegroundColor Cyan
    return @{ Nodes = $fNodes; Edges = $fEdges }
}

(function init() {
// ==== データ ====
var rawNodes = [$($nodesJson -join ',')];
var rawEdges = [$($edgesJson -join ',')];

// ==== 色設定 ====
var nodeColors = {
  form:'#3b82f6', module:'#8b5cf6', class:'#6366f1', method:'#22c55e', globalvar:'#f59e0b', designer:'#475569'
};
var edgeColors = {
  calls:'#60a5fa', file_calls:'#60a5fa', gv_read:'#a3e635', gv_write:'#f97316', navigate:'#f43f5e', contains:'#334155'
};
var nodeRadius = {file:18, class:14, method:8, gv:10};

// ==== 状態 ====
var nodes = rawNodes.map(function(d){ return Object.assign({},d); });
var edges = rawEdges.map(function(d){ return Object.assign({},d); });
var activeNodeGroups = new Set(['file','class','method','gv']);
var activeEdgeTypes = new Set(['calls','file_calls','gv_read','gv_write','navigate','contains']);
var searchText = '';
var selectedNode = null;

function getVisibleData() {
  var vn = nodes.filter(function(n){ return activeNodeGroups.has(n.group) && (searchText===''||n.label.toLowerCase().includes(searchText)); });
  var vnIds = new Set(vn.map(function(n){ return n.id; }));
  var ve = edges.filter(function(e){
    var sid = e.source.id||e.source, tid = e.target.id||e.target;
    return activeEdgeTypes.has(e.type) && vnIds.has(sid) && vnIds.has(tid);
  });
  return {nodes:vn, edges:ve};
}

var simulation = d3.forceSimulation()
  .force('link', d3.forceLink().id(function(d){return d.id;}).distance(function(d){return d.type==='contains'?40:100;}).strength(function(d){return d.type==='contains'?0.8:0.3;}))
  .force('charge', d3.forceManyBody().strength(-200))
  .force('center', d3.forceCenter(width/2, height/2))
  .force('collision', d3.forceCollide(25))
  .force('x', d3.forceX(width/2).strength(0.03))
  .force('y', d3.forceY(height/2).strength(0.03));

var linkG = g.append('g').attr('class','links');
var nodeG = g.append('g').attr('class','nodes');
var labelG = g.append('g').attr('class','labels');

function update() {
  var data = getVisibleData();
  var vn = data.nodes, ve = data.edges;

  // Links
  var link = linkG.selectAll('line').data(ve, function(d){ return (d.source.id||d.source)+'|'+(d.target.id||d.target)+'|'+d.type; });
  link.exit().remove();
  var linkEnter = link.enter().append('line')
    .attr('stroke', function(d){return edgeColors[d.type]||'#475569';})
    .attr('stroke-width', function(d){return d.type==='contains'?1:d.type==='file_calls'?2.5:1.5;})
    .attr('stroke-opacity', function(d){return d.type==='contains'?0.3:0.6;})
    .attr('stroke-dasharray', function(d){return d.type==='gv_read'?'4,3':d.type==='navigate'?'6,3':'none';})
    .attr('marker-end', function(d){return d.type!=='contains'?'url(#arrow-'+d.type+')':'';});
  var allLinks = linkEnter.merge(link);

  // Nodes
  var node = nodeG.selectAll('circle').data(vn, function(d){return d.id;});
  node.exit().remove();
  var nodeEnter = node.enter().append('circle')
    .attr('r', function(d){ return (nodeRadius[d.group]||10) * (d.focus?1.5:1); })
    .attr('fill', function(d){return nodeColors[d.type]||'#475569';})
    .attr('stroke', function(d){return d.focus?'#fbbf24':'#1e293b';})
    .attr('stroke-width', function(d){return d.focus?3:1.5;})
    .attr('cursor','pointer')
    .call(d3.drag().on('start',dragStart).on('drag',dragging).on('end',dragEnd))
    .on('mouseover', showTooltip).on('mouseout', hideTooltip)
    .on('click', function(ev,d){ selectNode(d); })
    .on('dblclick', function(ev,d){ focusOnNode(d); });
  var allNodes = nodeEnter.merge(node);

  // Labels
  var label = labelG.selectAll('text').data(vn, function(d){return d.id;});
  label.exit().remove();
  var labelEnter = label.enter().append('text')
    .attr('font-size', function(d){return d.group==='file'?11:d.group==='class'?10:8;})
    .attr('fill', '#94a3b8').attr('text-anchor','middle')
    .attr('dy', function(d){return -(nodeRadius[d.group]||10)-4;})
    .text(function(d){return d.label.replace('.vb','');});
  var allLabels = labelEnter.merge(label);

  simulation.nodes(vn);
  simulation.force('link').links(ve);
  simulation.alpha(0.5).restart();

  simulation.on('tick', function(){
    allLinks.attr('x1',function(d){return d.source.x;}).attr('y1',function(d){return d.source.y;}).attr('x2',function(d){return d.target.x;}).attr('y2',function(d){return d.target.y;});
    allNodes.attr('cx',function(d){return d.x;}).attr('cy',function(d){return d.y;});
    allLabels.attr('x',function(d){return d.x;}).attr('y',function(d){return d.y;});
  });

  document.getElementById('info-nodes').textContent = 'ノード: '+vn.length;
  document.getElementById('info-edges').textContent = 'エッジ: '+ve.length;
}

// ==== 選択&ハイライト ====
function selectNode(d) {
  selectedNode = selectedNode===d ? null : d;
  highlightConnected();
  updateNodeList();
}

function highlightConnected() {
  if (!selectedNode) {
    nodeG.selectAll('circle').attr('opacity',1);
    linkG.selectAll('line').attr('opacity', function(d){return d.type==='contains'?0.3:0.6;});
    labelG.selectAll('text').attr('opacity',1);
    return;
  }
  var connected = new Set([selectedNode.id]);
  var ve = getVisibleData().edges;
  ve.forEach(function(e){
    var sid = e.source.id||e.source, tid = e.target.id||e.target;
    if (sid===selectedNode.id) connected.add(tid);
    if (tid===selectedNode.id) connected.add(sid);
  });
  nodeG.selectAll('circle').attr('opacity', function(d){return connected.has(d.id)?1:0.1;});
  linkG.selectAll('line').attr('opacity', function(d){
    var sid=d.source.id||d.source, tid=d.target.id||d.target;
    return (sid===selectedNode.id||tid===selectedNode.id)?0.9:0.05;
  });
  labelG.selectAll('text').attr('opacity', function(d){return connected.has(d.id)?1:0.1;});
}

function focusOnNode(d) {
  var ve = getVisibleData().edges;
  var connected = new Set([d.id]);
  ve.forEach(function(e){
    var sid=e.source.id||e.source, tid=e.target.id||e.target;
    if(sid===d.id) connected.add(tid);
    if(tid===d.id) connected.add(sid);
  });
  var cnodes = nodes.filter(function(n){return connected.has(n.id);});
  if(cnodes.length===0)return;
  var xs=cnodes.map(function(n){return n.x;}),ys=cnodes.map(function(n){return n.y;});
  var x0=Math.min.apply(null,xs)-50,x1=Math.max.apply(null,xs)+50,y0=Math.min.apply(null,ys)-50,y1=Math.max.apply(null,ys)+50;
  var dx=x1-x0,dy=y1-y0;
  var scale=Math.min(width/dx,height/dy)*0.8;
  var tx=width/2-(x0+x1)/2*scale, ty=height/2-(y0+y1)/2*scale;
  svg.transition().duration(500).call(zoom.transform, d3.zoomIdentity.translate(tx,ty).scale(scale));
  selectedNode = d;
  highlightConnected();
}

// ==== ツールチップ ====
var tooltipEl = document.getElementById('tooltip');
function showTooltip(ev,d) {
  var data = getVisibleData();
  var incoming = data.edges.filter(function(e){return (e.target.id||e.target)===d.id;});
  var outgoing = data.edges.filter(function(e){return (e.source.id||e.source)===d.id;});
  var html = '<div class="tt-title">'+d.label+'</div>';
  html += '<div class="tt-row">種別: '+d.type+' ('+d.group+')</div>';
  if(incoming.length) html += '<div class="tt-row">\u2190 被参照: '+incoming.length+'件</div>';
  if(outgoing.length) html += '<div class="tt-row">\u2192 参照先: '+outgoing.length+'件</div>';
  var gvw = outgoing.filter(function(e){return e.type==='gv_write';}).length;
  var gvr = outgoing.filter(function(e){return e.type==='gv_read';}).length;
  if(gvw) html += '<div class="tt-row" style="color:#f97316">GV書込: '+gvw+'</div>';
  if(gvr) html += '<div class="tt-row" style="color:#a3e635">GV読取: '+gvr+'</div>';
  tooltipEl.innerHTML = html;
  tooltipEl.style.opacity = 1;
  tooltipEl.style.left = (ev.clientX-280+12)+'px';
  tooltipEl.style.top = (ev.clientY+12)+'px';
}
function hideTooltip() { tooltipEl.style.opacity = 0; }

// ==== ドラッグ ====
function dragStart(ev,d) { if(!ev.active)simulation.alphaTarget(0.3).restart(); d.fx=d.x; d.fy=d.y; }
function dragging(ev,d) { d.fx=ev.x; d.fy=ev.y; }
function dragEnd(ev,d) { if(!ev.active)simulation.alphaTarget(0); d.fx=null; d.fy=null; }

// ==== サイドバー: ノードリスト ====
function updateNodeList() {
  var vn = getVisibleData().nodes;
  var sorted = vn.slice().sort(function(a,b){ return (a.group+a.label).localeCompare(b.group+b.label); });
  var list = document.getElementById('nodeList');
  list.innerHTML = sorted.map(function(n){
    var color = nodeColors[n.type]||'#475569';
    var sel = selectedNode && selectedNode.id===n.id ? ' selected':'';
    return '<div class="node-item'+sel+'" data-id="'+n.id+'"><div class="node-dot" style="background:'+color+'"></div>'+n.label+'</div>';
  }).join('');
  list.querySelectorAll('.node-item').forEach(function(el){
    el.addEventListener('click',function(){
      var nd = nodes.find(function(n){return n.id===el.dataset.id;});
      if(nd) selectNode(nd);
    });
  });
}
