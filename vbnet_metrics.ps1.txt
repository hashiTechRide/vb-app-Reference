#Requires -Version 7.0
<#
.SYNOPSIS
    VB.NET 2008 CE プロジェクト メトリクス解析スクリプト
.DESCRIPTION
    対象: Visual Studio 2008 Compact Edition (VB.NET) プロジェクト
    出力: コンソール テキスト / ファイル出力
    依存: なし（PowerShell 7.x 標準機能のみ）
.PARAMETER Path
    解析対象のプロジェクトフォルダパス
.PARAMETER OutFile
    結果出力先ファイルパス（省略時はコンソール出力）
.EXAMPLE
    .\vbnet_metrics.ps1 -Path "C:\Projects\PickingApp"
    .\vbnet_metrics.ps1 -Path "C:\Projects\PickingApp" -OutFile "report.txt"
    .\vbnet_metrics.ps1 "C:\Projects\PickingApp" | Out-File report.txt -Encoding utf8
#>

[CmdletBinding()]
param(
    [Parameter(Mandatory, Position = 0)]
    [ValidateScript({ Test-Path $_ -PathType Container })]
    [string]$Path,

    [Parameter(Position = 1)]
    [string]$OutFile
)

$ErrorActionPreference = 'Stop'

# ============================================================
# データクラス
# ============================================================

class MethodInfo {
    [string]$Name
    [string]$FilePath
    [string]$ClassName
    [string]$AccessModifier
    [string]$ReturnType
    [string]$Params
    [int]$StartLine
    [int]$EndLine
    [int]$LineCount = 0
    [int]$BranchCount = 0
    [int]$MaxNestDepth = 0
    [int]$CyclomaticComplexity = 1
    [System.Collections.Generic.List[string]]$CallsTo = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$GVRead = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$GVWrite = [System.Collections.Generic.List[string]]::new()
    [bool]$HasDBAccess = $false
    [bool]$HasNetAccess = $false
}

class GlobalVarInfo {
    [string]$Name
    [string]$VarType
    [string]$DeclaredIn
    [string]$Modifier
    [string]$InitialValue = ''
    [System.Collections.Generic.List[string]]$SetLocations = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$ReadLocations = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.HashSet[string]]$RefFunctions = [System.Collections.Generic.HashSet[string]]::new()
    [System.Collections.Generic.HashSet[string]]$RefFiles = [System.Collections.Generic.HashSet[string]]::new()
}

class FormInfo {
    [string]$Name
    [string]$FilePath
    [System.Collections.Generic.List[string]]$Controls = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[hashtable]]$DynamicControls = [System.Collections.Generic.List[hashtable]]::new()
    [System.Collections.Generic.List[string]]$EventHandlers = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[string]]$NavigatesTo = [System.Collections.Generic.List[string]]::new()
}

class BranchInfo {
    [string]$FilePath
    [string]$MethodName
    [int]$LineNumber
    [string]$BranchType
    [string]$Condition
    [string]$Target = ''
    [int]$NestDepth
}

class FileMetrics {
    [string]$FilePath
    [string]$FileName
    [string]$FileType
    [int]$TotalLines = 0
    [int]$CodeLines = 0
    [int]$CommentLines = 0
    [int]$BlankLines = 0
    [int]$ClassCount = 0
    [int]$MethodCount = 0
    [int]$MaxMethodLines = 0
    [double]$AvgMethodLines = 0.0
    [int]$BranchCount = 0
    [int]$MaxNestDepth = 0
    [int]$GVRefCount = 0
}

# ============================================================
# パーサー
# ============================================================

class VBNetParser {
    [string]$ProjectPath
    [System.Collections.Generic.List[string]]$Files = [System.Collections.Generic.List[string]]::new()
    [System.Collections.Generic.List[FileMetrics]]$FileMetrics = [System.Collections.Generic.List[FileMetrics]]::new()
    [System.Collections.Generic.List[MethodInfo]]$Methods = [System.Collections.Generic.List[MethodInfo]]::new()
    [System.Collections.Generic.Dictionary[string, GlobalVarInfo]]$GlobalVars = [System.Collections.Generic.Dictionary[string, GlobalVarInfo]]::new()
    [System.Collections.Generic.Dictionary[string, FormInfo]]$Forms = [System.Collections.Generic.Dictionary[string, FormInfo]]::new()
    [System.Collections.Generic.List[BranchInfo]]$Branches = [System.Collections.Generic.List[BranchInfo]]::new()
    [hashtable]$AllLines = @{}

    # DB キーワード
    static [string[]]$DBKeywords = @(
        'sqlceconnection', 'sqlcecommand', 'sqlcedatareader', 'sqlcedataadapter',
        'sqlconnection', 'sqlcommand', 'sqldatareader', 'sqldataadapter',
        'executenonquery', 'executereader', 'executescalar',
        'oledbconnection', 'oledbcommand',
        'select ', 'insert ', 'update ', 'delete ', 'create table',
        '.open()', '.fill(', 'dataset', 'datatable'
    )
    # 通信キーワード
    static [string[]]$NetKeywords = @(
        'httpwebrequest', 'httpwebresponse', 'webclient', 'webrequest',
        'tcpclient', 'udpclient', 'socket', 'serialport', 'system.net',
        '.getresponse(', '.downloadstring(', '.uploadstring('
    )

    VBNetParser([string]$path) {
        $this.ProjectPath = $path
    }

    [void]DiscoverFiles() {
        Get-ChildItem -Path $this.ProjectPath -Filter '*.vb' -Recurse -File | ForEach-Object {
            $this.Files.Add($_.FullName)
        }
        $this.Files.Sort()
    }

    [string[]]ReadFile([string]$filePath) {
        foreach ($enc in @('utf-8', 'shift_jis', 'Default')) {
            try {
                $encoding = [System.Text.Encoding]::GetEncoding($enc)
                return [System.IO.File]::ReadAllLines($filePath, $encoding)
            }
            catch { continue }
        }
        # 最終フォールバック
        try { return [System.IO.File]::ReadAllLines($filePath) } catch { return @() }
    }

    [string]ClassifyFile([string]$filePath, [string]$content) {
        $fn = [System.IO.Path]::GetFileName($filePath).ToLower()
        if ($fn.EndsWith('.designer.vb')) { return 'Designer' }
        $cl = $content.ToLower()
        if ($cl.Contains('inherits system.windows.forms.form') -or $cl.Contains('inherits form')) { return 'Form' }
        if ($cl.Contains('inherits system.windows.forms.usercontrol') -or $cl.Contains('inherits usercontrol')) { return 'UserControl' }
        if ($cl -match '(?m)^\s*module\s+') { return 'Module' }
        return 'Class'
    }

    [void]CollectGlobalVars() {
        $reGV = [regex]::new('^\s*(Public|Friend)\s+(Shared\s+)?(Dim\s+|Const\s+)?(\w+)\s+As\s+(\w[\w\(\)]*)', 'IgnoreCase')
        $reClass = [regex]::new('^\s*(Public|Private|Friend|Protected)?\s*(Partial\s+)?(Class|Module|Structure)\s+(\w+)', 'IgnoreCase')
        $reEndClass = [regex]::new('^\s*End\s+(Class|Module|Structure)', 'IgnoreCase')

        foreach ($filePath in $this.Files) {
            $lines = $this.AllLines[$filePath]
            if (-not $lines) { continue }
            $fname = [System.IO.Path]::GetFileName($filePath)
            $inClass = $false
            $currentClass = ''

            foreach ($line in $lines) {
                $s = $line.Trim()
                $mc = $reClass.Match($s)
                if ($mc.Success) {
                    $inClass = $true
                    $currentClass = $mc.Groups[4].Value
                    continue
                }
                if ($reEndClass.IsMatch($s)) { $inClass = $false }

                $mg = $reGV.Match($s)
                if ($mg.Success -and $inClass) {
                    $vName = $mg.Groups[4].Value
                    $key = $vName.ToLower()
                    if (-not $this.GlobalVars.ContainsKey($key)) {
                        $gv = [GlobalVarInfo]::new()
                        $gv.Name = $vName
                        $gv.VarType = $mg.Groups[5].Value
                        $gv.DeclaredIn = "$fname ($currentClass)"
                        $gv.Modifier = $mg.Groups[1].Value
                        $initMatch = [regex]::Match($s, '=\s*(.+)$')
                        if ($initMatch.Success) { $gv.InitialValue = $initMatch.Groups[1].Value.Trim() }
                        $this.GlobalVars[$key] = $gv
                    }
                }
            }
        }
    }

    [void]Analyze() {
        foreach ($filePath in $this.Files) {
            $this.AllLines[$filePath] = $this.ReadFile($filePath)
        }
        $this.CollectGlobalVars()

        foreach ($filePath in $this.Files) {
            $lines = $this.AllLines[$filePath]
            $content = $lines -join "`n"
            $fname = [System.IO.Path]::GetFileName($filePath)
            $ftype = $this.ClassifyFile($filePath, $content)

            $fm = [FileMetrics]::new()
            $fm.FilePath = $filePath
            $fm.FileName = $fname
            $fm.FileType = $ftype
            $this.CalcLineMetrics($lines, $fm)

            if ($ftype -ne 'Designer') {
                $this.ParseMethods($filePath, $lines, $fm)
                $this.ScanGVRefs($filePath, $lines, $fm)
                if ($ftype -in @('Form', 'UserControl')) {
                    $this.ParseForm($filePath, $lines, $content)
                }
            }
            $this.FileMetrics.Add($fm)
        }
    }

    [void]CalcLineMetrics([string[]]$lines, [FileMetrics]$fm) {
        $fm.TotalLines = $lines.Count
        foreach ($line in $lines) {
            $s = $line.Trim()
            if ([string]::IsNullOrEmpty($s)) { $fm.BlankLines++ }
            elseif ($s.StartsWith("'") -or $s.ToUpper().StartsWith('REM ')) { $fm.CommentLines++ }
            else { $fm.CodeLines++ }
        }
    }

    [void]ParseMethods([string]$filePath, [string[]]$lines, [FileMetrics]$fm) {
        $fname = [System.IO.Path]::GetFileName($filePath)
        $reClass = [regex]::new('^\s*(Public|Private|Friend|Protected)?\s*(Partial\s+)?(Class|Module|Structure)\s+(\w+)', 'IgnoreCase')
        $reMethod = [regex]::new('^\s*(Public|Private|Protected|Friend)?\s*(Shared\s+)?(Overrides\s+)?(Sub|Function)\s+(\w+)\s*\(([^)]*)\)(?:\s+As\s+(\w+))?', 'IgnoreCase')
        $reEndBlock = [regex]::new('^\s*End\s+(Sub|Function|Class|Module|Structure|Property|If|Select|While|For|Try|With|Using|SyncLock)', 'IgnoreCase')
        $reIf = [regex]::new('^\s*If\s+(.+?)\s+Then\s*$', 'IgnoreCase')
        $reElseIf = [regex]::new('^\s*ElseIf\s+(.+?)\s+Then', 'IgnoreCase')
        $reSelect = [regex]::new('^\s*Select\s+Case\s+(.+)', 'IgnoreCase')
        $reCase = [regex]::new('^\s*Case\s+(.+)', 'IgnoreCase')
        $reFor = [regex]::new('^\s*For\s+', 'IgnoreCase')
        $reForEach = [regex]::new('^\s*For\s+Each\s+', 'IgnoreCase')
        $reWhile = [regex]::new('^\s*(?:While|Do\s+While|Do\s+Until)\s+', 'IgnoreCase')
        $reTry = [regex]::new('^\s*Try\s*$', 'IgnoreCase')
        $reCatch = [regex]::new('^\s*Catch\b', 'IgnoreCase')
        $reCtrlProp = [regex]::new('(\w+)\.(Text|Visible|Enabled|BackColor|ForeColor|Caption|Name|DataSource|Columns)\s*=\s*(.+)', 'IgnoreCase')

        $currentClass = ''
        [System.Collections.Generic.List[object]]$methodStack = [System.Collections.Generic.List[object]]::new()
        $nest = 0

        for ($i = 0; $i -lt $lines.Count; $i++) {
            $code = ($lines[$i] -replace "'.*$", '').Trim()
            if ([string]::IsNullOrEmpty($code)) { continue }
            $ln = $i + 1

            $mc = $reClass.Match($code)
            if ($mc.Success) {
                $currentClass = $mc.Groups[4].Value
                $fm.ClassCount++
                continue
            }

            $mm = $reMethod.Match($code)
            if ($mm.Success) {
                $mi = [MethodInfo]::new()
                $mi.Name = $mm.Groups[5].Value
                $mi.FilePath = $fname
                $mi.ClassName = $currentClass
                $mi.AccessModifier = if ($mm.Groups[1].Value) { $mm.Groups[1].Value } else { 'Private' }
                $mi.ReturnType = if ($mm.Groups[7].Value) { $mm.Groups[7].Value } elseif ($mm.Groups[4].Value -ieq 'sub') { 'Void' } else { '?' }
                $mi.Params = ($mm.Groups[6].Value ?? '').Trim()
                $mi.StartLine = $ln
                $mi.EndLine = $ln
                $methodStack.Add(@{ Method = $mi; BaseNest = $nest })
                $nest++
                $fm.MethodCount++
                continue
            }

            # ---- 分岐解析 ----
            $currentMethod = if ($methodStack.Count -gt 0) { $methodStack[-1].Method } else { $null }

            $mIf = $reIf.Match($code)
            if ($mIf.Success) {
                $nest++
                if ($currentMethod) {
                    $cond = $mIf.Groups[1].Value
                    $currentMethod.BranchCount++
                    $currentMethod.CyclomaticComplexity++
                    $fm.BranchCount++
                    $bi = [BranchInfo]::new()
                    $bi.FilePath = $fname; $bi.MethodName = $currentMethod.Name
                    $bi.LineNumber = $ln; $bi.BranchType = 'If'
                    $bi.Condition = $cond.Substring(0, [Math]::Min($cond.Length, 120))
                    $bi.NestDepth = $nest
                    $this.Branches.Add($bi)
                    $extra = [regex]::Matches($cond, '\b(And|Or|AndAlso|OrElse)\b', 'IgnoreCase').Count
                    $currentMethod.CyclomaticComplexity += $extra
                }
            }
            elseif ($reElseIf.IsMatch($code)) {
                if ($currentMethod) {
                    $cond = $reElseIf.Match($code).Groups[1].Value
                    $currentMethod.BranchCount++
                    $currentMethod.CyclomaticComplexity++
                    $fm.BranchCount++
                    $bi = [BranchInfo]::new()
                    $bi.FilePath = $fname; $bi.MethodName = $currentMethod.Name
                    $bi.LineNumber = $ln; $bi.BranchType = 'ElseIf'
                    $bi.Condition = $cond.Substring(0, [Math]::Min($cond.Length, 120))
                    $bi.NestDepth = $nest
                    $this.Branches.Add($bi)
                }
            }
            elseif ($reSelect.IsMatch($code)) {
                $nest++
                if ($currentMethod) {
                    $cond = $reSelect.Match($code).Groups[1].Value
                    $currentMethod.BranchCount++
                    $currentMethod.CyclomaticComplexity++
                    $fm.BranchCount++
                    $bi = [BranchInfo]::new()
                    $bi.FilePath = $fname; $bi.MethodName = $currentMethod.Name
                    $bi.LineNumber = $ln; $bi.BranchType = 'Select Case'
                    $bi.Condition = $cond.Substring(0, [Math]::Min($cond.Length, 120))
                    $bi.NestDepth = $nest
                    $this.Branches.Add($bi)
                }
            }
            elseif ($reCase.IsMatch($code) -and -not $code.Trim().ToLower().StartsWith('select')) {
                if ($currentMethod -and -not $code.Trim().ToLower().StartsWith('case else')) {
                    $currentMethod.CyclomaticComplexity++
                }
            }
            elseif ($reFor.IsMatch($code) -or $reForEach.IsMatch($code)) {
                $nest++
                if ($currentMethod) { $currentMethod.CyclomaticComplexity++ }
            }
            elseif ($reWhile.IsMatch($code)) {
                $nest++
                if ($currentMethod) { $currentMethod.CyclomaticComplexity++ }
            }
            elseif ($reTry.IsMatch($code)) { $nest++ }
            elseif ($reCatch.IsMatch($code)) {
                if ($currentMethod) { $currentMethod.CyclomaticComplexity++ }
            }

            # ネスト深度追跡
            if ($methodStack.Count -gt 0) {
                $baseNest = $methodStack[-1].BaseNest
                $rel = $nest - $baseNest - 1
                if ($rel -gt $currentMethod.MaxNestDepth) {
                    $currentMethod.MaxNestDepth = $rel
                }
            }

            # コントロール動的変更 → 分岐ターゲット
            if ($currentMethod) {
                $cp = $reCtrlProp.Match($code)
                if ($cp.Success) {
                    for ($bi_idx = $this.Branches.Count - 1; $bi_idx -ge 0; $bi_idx--) {
                        $b = $this.Branches[$bi_idx]
                        if ($b.FilePath -eq $fname -and $b.MethodName -eq $currentMethod.Name -and [string]::IsNullOrEmpty($b.Target)) {
                            $b.Target = "$($cp.Groups[1].Value).$($cp.Groups[2].Value)"
                            break
                        }
                    }
                }
            }

            # DB / Network
            if ($currentMethod) {
                $cl = $code.ToLower()
                foreach ($kw in [VBNetParser]::DBKeywords) {
                    if ($cl.Contains($kw)) { $currentMethod.HasDBAccess = $true; break }
                }
                foreach ($kw in [VBNetParser]::NetKeywords) {
                    if ($cl.Contains($kw)) { $currentMethod.HasNetAccess = $true; break }
                }
            }

            # End block
            $em = $reEndBlock.Match($code)
            if ($em.Success) {
                $bt = $em.Groups[1].Value.ToLower()
                if ($bt -in @('sub', 'function') -and $methodStack.Count -gt 0) {
                    $entry = $methodStack[-1]
                    $mi = $entry.Method
                    $methodStack.RemoveAt($methodStack.Count - 1)
                    $mi.EndLine = $ln
                    $mi.LineCount = $mi.EndLine - $mi.StartLine + 1
                    $this.ScanMethodBody($filePath, $lines, $mi)
                    $this.Methods.Add($mi)
                }
                $nest = [Math]::Max(0, $nest - 1)
            }
        }

        $mlens = $this.Methods | Where-Object { $_.FilePath -eq $fname } | ForEach-Object { $_.LineCount }
        if ($mlens) {
            $fm.MaxMethodLines = ($mlens | Measure-Object -Maximum).Maximum
            $fm.AvgMethodLines = ($mlens | Measure-Object -Average).Average
        }
        $nests = $this.Methods | Where-Object { $_.FilePath -eq $fname } | ForEach-Object { $_.MaxNestDepth }
        if ($nests) { $fm.MaxNestDepth = ($nests | Measure-Object -Maximum).Maximum }
    }

    [void]ScanMethodBody([string]$filePath, [string[]]$lines, [MethodInfo]$mi) {
        $skipCalls = [System.Collections.Generic.HashSet[string]]::new(
            [string[]]@('if','while','for','select','case','catch','ctype','cstr','cint','cdbl',
                        'cbool','directcast','trycast','typeof','gettype','nothing','not','and','or'),
            [System.StringComparer]::OrdinalIgnoreCase
        )
        $startIdx = [Math]::Max(0, $mi.StartLine - 1)
        $endIdx = [Math]::Min($mi.EndLine, $lines.Count)

        for ($li = $startIdx; $li -lt $endIdx; $li++) {
            $code = ($lines[$li] -replace "'.*$", '').Trim()
            if ([string]::IsNullOrEmpty($code)) { continue }
            $loc = "$($mi.FilePath):$($li + 1)"

            foreach ($key in $this.GlobalVars.Keys) {
                $gv = $this.GlobalVars[$key]
                if ($code -match "\b$([regex]::Escape($gv.Name))\b") {
                    if ($code -match "\b$([regex]::Escape($gv.Name))\s*=") {
                        if (-not $mi.GVWrite.Contains($gv.Name)) { $mi.GVWrite.Add($gv.Name) }
                        if (-not $gv.SetLocations.Contains($loc)) { $gv.SetLocations.Add($loc) }
                    }
                    else {
                        if (-not $mi.GVRead.Contains($gv.Name)) { $mi.GVRead.Add($gv.Name) }
                        if (-not $gv.ReadLocations.Contains($loc)) { $gv.ReadLocations.Add($loc) }
                    }
                    [void]$gv.RefFunctions.Add("$($mi.ClassName).$($mi.Name)")
                    [void]$gv.RefFiles.Add($mi.FilePath)
                }
            }

            $callMatches = [regex]::Matches($code, '(\w+\.\w+|\w+)\s*\(')
            foreach ($cm in $callMatches) {
                $call = $cm.Groups[1].Value
                if (-not $skipCalls.Contains($call) -and -not $mi.CallsTo.Contains($call)) {
                    $mi.CallsTo.Add($call)
                }
            }
        }
    }

    [void]ScanGVRefs([string]$filePath, [string[]]$lines, [FileMetrics]$fm) {
        $count = 0
        foreach ($line in $lines) {
            $code = ($line -replace "'.*$", '').Trim()
            foreach ($gv in $this.GlobalVars.Values) {
                if ($code -match "\b$([regex]::Escape($gv.Name))\b") { $count++ }
            }
        }
        $fm.GVRefCount = $count
    }

    [void]ParseForm([string]$filePath, [string[]]$lines, [string]$content) {
        $fname = [System.IO.Path]::GetFileName($filePath)
        $formName = [System.IO.Path]::GetFileNameWithoutExtension($fname)
        $fi = [FormInfo]::new()
        $fi.Name = $formName
        $fi.FilePath = $fname

        $reCtrl1 = [regex]::new('(?:Me\.)?(\w+)\s*=\s*New\s+System\.Windows\.Forms\.(\w+)', 'IgnoreCase')
        $reCtrl2 = [regex]::new('Friend\s+WithEvents\s+(\w+)\s+As\s+(?:System\.Windows\.Forms\.)?(\w+)', 'IgnoreCase')
        $reHandler = [regex]::new('(?:Private|Public|Protected)\s+Sub\s+(\w+_\w+)\s*\(', 'IgnoreCase')
        $reShow = [regex]::new('(\w+)\.(Show|ShowDialog)\b', 'IgnoreCase')
        $reCtrlProp = [regex]::new('(\w+)\.(Text|Visible|Enabled|BackColor|ForeColor|Caption|Name|DataSource|Columns)\s*=\s*(.+)', 'IgnoreCase')

        $allLines = [System.Collections.Generic.List[string]]::new($lines)
        $designerPath = $filePath -replace '\.vb$', '.Designer.vb'
        if (Test-Path $designerPath) {
            $allLines.AddRange($this.ReadFile($designerPath))
        }

        $ctrls = [System.Collections.Generic.HashSet[string]]::new()
        foreach ($line in $allLines) {
            foreach ($re in @($reCtrl1, $reCtrl2)) {
                $m = $re.Match($line)
                if ($m.Success) { [void]$ctrls.Add("$($m.Groups[1].Value) ($($m.Groups[2].Value))") }
            }
        }
        $fi.Controls.AddRange([string[]]($ctrls | Sort-Object))

        foreach ($line in $lines) {
            $m = $reHandler.Match($line.Trim())
            if ($m.Success) { $fi.EventHandlers.Add($m.Groups[1].Value) }
            $m = $reShow.Match($line)
            if ($m.Success -and -not $fi.NavigatesTo.Contains($m.Groups[1].Value)) {
                $fi.NavigatesTo.Add($m.Groups[1].Value)
            }
        }

        for ($li = 0; $li -lt $lines.Count; $li++) {
            $code = ($lines[$li] -replace "'.*$", '').Trim()
            $m = $reCtrlProp.Match($code)
            if ($m.Success) {
                $val = $m.Groups[3].Value.Trim()
                $fi.DynamicControls.Add(@{
                    Control  = $m.Groups[1].Value
                    Property = $m.Groups[2].Value
                    Value    = $val.Substring(0, [Math]::Min($val.Length, 60))
                    Line     = $li + 1
                })
            }
        }

        $this.Forms[$formName] = $fi
    }
}

class TextReport {
    [VBNetParser]$P
    [System.Text.StringBuilder]$Buf = [System.Text.StringBuilder]::new()
    static [string]$SepThick = '=' * 90
    static [string]$SepThin = '-' * 90

    TextReport([VBNetParser]$parser) { $this.P = $parser }

    [void]W([string]$text) { [void]$this.Buf.AppendLine($text) }
    [void]W() { [void]$this.Buf.AppendLine('') }

    [void]Header([string]$title, [int]$level) {
        $this.W()
        switch ($level) {
            1 { $this.W([TextReport]::SepThick); $this.W("  $title"); $this.W([TextReport]::SepThick) }
            2 { $this.W([TextReport]::SepThin); $this.W("  $title"); $this.W([TextReport]::SepThin) }
            default { $this.W("  --- $title ---") }
        }
    }

    [void]Table([string[]]$headers, [object[][]]$rows, [int[]]$widths) {
        $fmtRow = {
            param([object[]]$vals, [int[]]$ws)
            $parts = for ($i = 0; $i -lt $vals.Count; $i++) {
                $w = if ($i -lt $ws.Count) { $ws[$i] } else { 20 }
                "$($vals[$i])".PadRight($w)
            }
            "  $($parts -join ' | ')"
        }
        $this.W((& $fmtRow $headers $widths))
        $divider = ($widths | ForEach-Object { '-' * $_ }) -join '+'
        $this.W("  $divider")
        foreach ($row in $rows) {
            $this.W((& $fmtRow $row $widths))
        }
    }

    [string]Generate() {
        $this.GenTitle()
        $this.GenSummary()
        $this.GenFileMetrics()
        $this.GenComplexityRanking()
        $this.GenGlobalVars()
        $this.GenBranchMap()
        $this.GenFormMap()
        $this.GenMethodDetail()
        $this.GenDependencyMatrix()
        return $this.Buf.ToString()
    }

    [void]GenTitle() {
        $this.W([TextReport]::SepThick)
        $this.W('  VB.NET 2008 CE プロジェクト メトリクス解析レポート')
        $this.W([TextReport]::SepThick)
        $this.W("  解析日時 : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')")
        $this.W("  対象     : $($this.P.ProjectPath)")
        $this.W("  ファイル数: $($this.P.Files.Count)")
    }

    [void]GenSummary() {
        $this.Header('1. サマリー', 1)
        $nd = $this.P.FileMetrics | Where-Object { $_.FileType -ne 'Designer' }
        $total = ($nd | Measure-Object -Property TotalLines -Sum).Sum
        $code = ($nd | Measure-Object -Property CodeLines -Sum).Sum
        $comment = ($nd | Measure-Object -Property CommentLines -Sum).Sum
        $blank = ($nd | Measure-Object -Property BlankLines -Sum).Sum
        $nm = $this.P.Methods.Count
        $avgCC = if ($nm -gt 0) { ($this.P.Methods | Measure-Object -Property CyclomaticComplexity -Average).Average } else { 0 }
        $maxCC = if ($nm -gt 0) { ($this.P.Methods | Measure-Object -Property CyclomaticComplexity -Maximum).Maximum } else { 0 }
        $maxNest = if ($nm -gt 0) { ($this.P.Methods | Measure-Object -Property MaxNestDepth -Maximum).Maximum } else { 0 }
        $hiRiskM = ($this.P.Methods | Where-Object { $_.CyclomaticComplexity -gt 10 }).Count
        $bigM = ($this.P.Methods | Where-Object { $_.LineCount -gt 100 }).Count
        $gvCount = $this.P.GlobalVars.Count
        $hiRiskGV = ($this.P.GlobalVars.Values | Where-Object { $_.RefFiles.Count -ge 3 }).Count
        $formCount = ($nd | Where-Object { $_.FileType -eq 'Form' }).Count
        $modCount = ($nd | Where-Object { $_.FileType -eq 'Module' }).Count
        $classCount = ($nd | Measure-Object -Property ClassCount -Sum).Sum
        $branchTotal = ($nd | Measure-Object -Property BranchCount -Sum).Sum
        $commentRate = if ($code -gt 0) { '{0:F1}%' -f ($comment / $code * 100) } else { '0%' }

        $this.W()
        $this.W('  [コード量]')
        foreach ($pair in @(
            @('ファイル数 (Designer除く)', @($nd).Count),
            @('総行数', $total), @('コード行', $code),
            @('コメント行', $comment), @('空白行', $blank),
            @('コメント率', $commentRate),
            @('フォーム数', $formCount), @('モジュール数', $modCount),
            @('クラス数', $classCount), @('メソッド数', $nm)
        )) { $this.W("    $($pair[0].PadRight(32)): $($pair[1])") }

        $this.W()
        $this.W('  [複雑度]')
        foreach ($pair in @(
            @('平均循環的複雑度', '{0:F1}' -f $avgCC),
            @('最大循環的複雑度', $maxCC), @('最大ネスト深度', $maxNest),
            @('高リスクメソッド (CC>10)', $hiRiskM),
            @('巨大メソッド (100行超)', $bigM), @('総分岐数', $branchTotal)
        )) { $this.W("    $($pair[0].PadRight(32)): $($pair[1])") }

        $this.W()
        $this.W('  [グローバル変数]')
        $gvRefTotal = ($this.P.GlobalVars.Values | ForEach-Object { $_.ReadLocations.Count + $_.SetLocations.Count } | Measure-Object -Sum).Sum
        foreach ($pair in @(
            @('グローバル変数数', $gvCount),
            @('高リスク (3ファイル以上参照)', $hiRiskGV),
            @('参照箇所 総数', $gvRefTotal)
        )) { $this.W("    $($pair[0].PadRight(32)): $($pair[1])") }

        $this.W()
        $this.W('  [フォーム共通化]')
        $dynTotal = ($this.P.Forms.Values | ForEach-Object { $_.DynamicControls.Count } | Measure-Object -Sum).Sum
        $dynForms = ($this.P.Forms.Values | Where-Object { $_.DynamicControls.Count -gt 0 }).Count
        foreach ($pair in @(
            @('フォーム数', $this.P.Forms.Count),
            @('動的コントロール変更 総数', $dynTotal),
            @('切替有りフォーム数', $dynForms)
        )) { $this.W("    $($pair[0].PadRight(32)): $($pair[1])") }

        $this.W()
        $this.W('  [リスク警告]')
        $warned = $false
        if ($hiRiskM -gt 0) { $this.W("    !! 循環的複雑度 > 10 のメソッドが $hiRiskM 件。分割を検討。"); $warned = $true }
        if ($hiRiskGV -gt 0) { $this.W("    !! 3ファイル以上参照のグローバル変数が $hiRiskGV 件。引数化・クラス化を推奨。"); $warned = $true }
        if ($bigM -gt 0) { $this.W("    !! 100行超のメソッドが $bigM 件。"); $warned = $true }
        $dynHigh = @($this.P.Forms.Values | Where-Object { $_.DynamicControls.Count -gt 5 })
        if ($dynHigh.Count -gt 0) { $this.W("    !! 動的変更 > 5 のフォームが $($dynHigh.Count) 件。フォーム分離を検討。"); $warned = $true }
        if (-not $warned) { $this.W('    重大なリスクは検出されませんでした。') }
    }

    [void]GenFileMetrics() {
        $this.Header('2. ファイル別メトリクス', 1)
        $nd = $this.P.FileMetrics | Where-Object { $_.FileType -ne 'Designer' } | Sort-Object CodeLines -Descending
        $headers = @('ファイル名','種別','コード行','コメント','メソッド数','最大行','分岐数','最大ネスト','GV参照')
        $rows = foreach ($f in $nd) {
            ,@($f.FileName, $f.FileType, $f.CodeLines, $f.CommentLines, $f.MethodCount,
               $f.MaxMethodLines, $f.BranchCount, $f.MaxNestDepth, $f.GVRefCount)
        }
        $this.Table($headers, $rows, @(24,12,8,8,8,8,8,8,8))
    }

    [void]GenComplexityRanking() {
        $this.Header('3. 複雑度ランキング（リファクタリング優先度 Top30）', 1)
        $ranked = $this.P.Methods | ForEach-Object {
            $s = $_.CyclomaticComplexity * 3 + $_.LineCount * 0.1 + $_.MaxNestDepth * 5 + ($_.GVRead.Count + $_.GVWrite.Count) * 4
            [PSCustomObject]@{ M = $_; Score = $s }
        } | Sort-Object Score -Descending | Select-Object -First 30

        $headers = @('#','ファイル','クラス','メソッド','行数','CC','ネスト','GV数','スコア','推奨')
        $idx = 0
        $rows = foreach ($r in $ranked) {
            $idx++
            $m = $r.M; $s = $r.Score
            $gv = $m.GVRead.Count + $m.GVWrite.Count
            $act = if ($s -gt 50) { '要分割+GV除去' } elseif ($s -gt 30) { '分割推奨' } elseif ($s -gt 15) { 'リファクタ検討' } else { '許容範囲' }
            ,@($idx, $m.FilePath, $m.ClassName, $m.Name, $m.LineCount, $m.CyclomaticComplexity, $m.MaxNestDepth, $gv, ('{0:F0}' -f $s), $act)
        }
        $this.Table($headers, $rows, @(3,18,14,22,5,4,5,4,6,16))
    }

    [void]GenGlobalVars() {
        $this.Header('4. グローバル変数 影響範囲 ★', 1)
        if ($this.P.GlobalVars.Count -eq 0) {
            $this.W('  グローバル変数は検出されませんでした。')
            return
        }

        $sorted = $this.P.GlobalVars.Values | Sort-Object { $_.RefFiles.Count } -Descending
        foreach ($gv in $sorted) {
            $fc = $gv.RefFiles.Count
            $risk = if ($fc -ge 4) { '高' } elseif ($fc -ge 2) { '中' } else { '低' }
            $marker = switch ($risk) { '高' { ' ★★★' } '中' { ' ★★' } default { '' } }

            $this.W()
            $this.W("  [$($gv.Name)]$marker  リスク: $risk")
            $this.W("    型          : $($gv.VarType)")
            $this.W("    宣言場所    : $($gv.DeclaredIn)")
            $this.W("    修飾子      : $($gv.Modifier)")
            $this.W("    初期値      : $(if ($gv.InitialValue) { $gv.InitialValue } else { '(なし)' })")
            $this.W("    参照ファイル数: $fc")
            $this.W("    セット箇所数 : $($gv.SetLocations.Count)")
            $this.W("    参照箇所数  : $($gv.ReadLocations.Count)")

            if ($gv.SetLocations.Count -gt 0) {
                $this.W('    セット箇所  :')
                foreach ($loc in ($gv.SetLocations | Select-Object -First 15)) { $this.W("      - $loc") }
            }
            if ($gv.ReadLocations.Count -gt 0) {
                $this.W('    参照箇所    :')
                foreach ($loc in ($gv.ReadLocations | Select-Object -First 15)) { $this.W("      - $loc") }
            }
            if ($gv.RefFunctions.Count -gt 0) {
                $this.W('    参照関数    :')
                foreach ($fn in ($gv.RefFunctions | Sort-Object | Select-Object -First 15)) { $this.W("      - $fn") }
            }
        }

        $this.W()
        $this.Header('グローバル変数 リスク集計', 3)
        $hi = @($this.P.GlobalVars.Values | Where-Object { $_.RefFiles.Count -ge 4 }).Count
        $mid = @($this.P.GlobalVars.Values | Where-Object { $_.RefFiles.Count -ge 2 -and $_.RefFiles.Count -lt 4 }).Count
        $lo = @($this.P.GlobalVars.Values | Where-Object { $_.RefFiles.Count -lt 2 }).Count
        $this.W("    高: $hi 件")
        $this.W("    中: $mid 件")
        $this.W("    低: $lo 件")
    }

    [void]GenBranchMap() {
        $this.Header('5. 条件分岐マップ', 1)
        if ($this.P.Branches.Count -eq 0) { $this.W('  分岐は検出されませんでした。'); return }

        $byFile = $this.P.Branches | Group-Object FilePath | Sort-Object Name
        foreach ($grp in $byFile) {
            $this.Header("[$($grp.Name)]", 3)
            $headers = @('行','種別','メソッド','条件','対象ctrl','ネスト')
            $rows = foreach ($b in ($grp.Group | Sort-Object LineNumber)) {
                $cond = $b.Condition
                if ($cond.Length -gt 50) { $cond = $cond.Substring(0, 50) }
                ,@($b.LineNumber, $b.BranchType, $b.MethodName, $cond, $(if ($b.Target) { $b.Target } else { '-' }), $b.NestDepth)
            }
            $this.Table($headers, $rows, @(5,12,22,50,18,6))
        }

        $ctrlBranches = $this.P.Branches | Where-Object { $_.Target }
        if ($ctrlBranches) {
            $this.W()
            $this.Header('コントロール動的切替の分岐（注目）', 3)
            $headers = @('ファイル','メソッド','行','条件','対象')
            $rows = foreach ($b in $ctrlBranches) {
                $cond = $b.Condition; if ($cond.Length -gt 40) { $cond = $cond.Substring(0, 40) }
                ,@($b.FilePath, $b.MethodName, $b.LineNumber, $cond, $b.Target)
            }
            $this.Table($headers, $rows, @(18,22,5,40,20))
        }
    }

    [void]GenFormMap() {
        $this.Header('6. フォーム共通化・動的コントロールマップ', 1)
        if ($this.P.Forms.Count -eq 0) { $this.W('  フォームは検出されませんでした。'); return }

        $sorted = $this.P.Forms.Values | Sort-Object { $_.DynamicControls.Count } -Descending
        foreach ($fi in $sorted) {
            $dyn = $fi.DynamicControls.Count
            $marker = if ($dyn -gt 5) { ' ★★ フォーム分離候補' } elseif ($dyn -gt 0) { ' ★ 要確認' } else { '' }
            $this.W()
            $this.W("  [$($fi.Name)]$marker")
            $this.W("    コントロール数     : $($fi.Controls.Count)")
            $this.W("    イベントハンドラ数 : $($fi.EventHandlers.Count)")
            $this.W("    動的コントロール変更: $dyn")
            $this.W("    遷移先             : $(if ($fi.NavigatesTo.Count -gt 0) { $fi.NavigatesTo -join ', ' } else { '-' })")

            if ($fi.DynamicControls.Count -gt 0) {
                $this.W('    動的変更詳細:')
                foreach ($d in ($fi.DynamicControls | Select-Object -First 20)) {
                    $this.W("      L$($d.Line.ToString().PadLeft(4))  $($d.Control).$($d.Property) = $($d.Value)")
                }
            }
            if ($fi.Controls.Count -gt 0) {
                $this.W('    コントロール一覧:')
                foreach ($ctrl in ($fi.Controls | Select-Object -First 20)) { $this.W("      - $ctrl") }
            }
        }
    }

    [void]GenMethodDetail() {
        $this.Header('7. メソッド詳細一覧', 1)
        $headers = @('ファイル','クラス','メソッド','修飾子','行数','CC','ネスト','GV(R)','GV(W)','DB','通信')
        $sorted = $this.P.Methods | Sort-Object FilePath, StartLine
        $rows = foreach ($m in $sorted) {
            $gvr = if ($m.GVRead.Count -gt 0) { ($m.GVRead -join ',').Substring(0, [Math]::Min(($m.GVRead -join ',').Length, 25)) } else { '-' }
            $gvw = if ($m.GVWrite.Count -gt 0) { ($m.GVWrite -join ',').Substring(0, [Math]::Min(($m.GVWrite -join ',').Length, 25)) } else { '-' }
            ,@($m.FilePath, $m.ClassName, $m.Name, $m.AccessModifier, $m.LineCount,
               $m.CyclomaticComplexity, $m.MaxNestDepth, $gvr, $gvw,
               $(if ($m.HasDBAccess) {'Y'} else {'-'}), $(if ($m.HasNetAccess) {'Y'} else {'-'}))
        }
        $this.Table($headers, $rows, @(18,14,22,8,5,4,5,25,25,3,3))
    }

    [void]GenDependencyMatrix() {
        $this.Header('8. ファイル間依存関係マトリクス', 1)
        $this.W('  行 = 呼出元, 列 = 呼出先. 数値 = メソッド呼出回数.')
        $this.W()

        $files = @($this.P.Methods | ForEach-Object { $_.FilePath } | Sort-Object -Unique)
        if ($files.Count -eq 0) { $this.W('  データなし'); return }

        $idx = @{}
        for ($i = 0; $i -lt $files.Count; $i++) { $idx[$files[$i]] = $i }
        $n = $files.Count
        $mat = for ($i = 0; $i -lt $n; $i++) { , ([int[]]::new($n)) }

        $c2f = @{}
        foreach ($m in $this.P.Methods) { $c2f[$m.ClassName.ToLower()] = $m.FilePath }

        foreach ($m in $this.P.Methods) {
            $si = $idx[$m.FilePath]
            foreach ($call in $m.CallsTo) {
                $parts = $call.Split('.')
                $tc = if ($parts.Count -gt 1) { $parts[0].ToLower() } else { $m.ClassName.ToLower() }
                $tf = if ($c2f.ContainsKey($tc)) { $c2f[$tc] } else { $m.FilePath }
                if ($idx.ContainsKey($tf)) { $mat[$si][$idx[$tf]]++ }
            }
        }

        $short = $files | ForEach-Object { $_.Replace('.vb','').Substring(0, [Math]::Min($_.Replace('.vb','').Length, 12)) }
        $cw = [Math]::Max(($short | ForEach-Object { $_.Length } | Measure-Object -Maximum).Maximum + 1, 5)

        $hdr = (' ' * ($cw + 2)) + (($short | ForEach-Object { $_.PadLeft($cw) }) -join '')
        $this.W("  $hdr")
        $this.W("  $('-' * $hdr.Length)")

        for ($i = 0; $i -lt $n; $i++) {
            $rowStr = $short[$i].PadRight($cw) + ' |'
            for ($j = 0; $j -lt $n; $j++) {
                $v = $mat[$i][$j]
                $rowStr += $(if ($v -gt 0) { "$v" } else { '.' }).PadLeft($cw)
            }
            $this.W("  $rowStr")
        }

        $this.W()
        $this.Header('ファイル間依存 サマリー（自ファイル除く）', 3)
        $deps = [System.Collections.Generic.List[object]]::new()
        for ($i = 0; $i -lt $n; $i++) {
            for ($j = 0; $j -lt $n; $j++) {
                if ($i -ne $j -and $mat[$i][$j] -gt 0) {
                    $deps.Add(@{ From = $files[$i]; To = $files[$j]; Count = $mat[$i][$j] })
                }
            }
        }
        if ($deps.Count -gt 0) {
            $depsSorted = $deps | Sort-Object { $_.Count } -Descending | Select-Object -First 20
            $headers = @('呼出元','呼出先','回数')
            $rows = foreach ($d in $depsSorted) { ,@($d.From, $d.To, $d.Count) }
            $this.Table($headers, $rows, @(22,22,6))
        }
        else { $this.W('  ファイル間の依存はありません。') }
    }
}

# ============================================================
# メイン処理
# ============================================================

$resolvedPath = Resolve-Path $Path

Write-Host '解析中...' -ForegroundColor Cyan
$parser = [VBNetParser]::new($resolvedPath.Path)
$parser.DiscoverFiles()

if ($parser.Files.Count -eq 0) {
    Write-Error '.vb ファイルが見つかりませんでした。'
    exit 1
}
Write-Host "  ファイル数: $($parser.Files.Count)" -ForegroundColor Gray
$parser.Analyze()
Write-Host "  メソッド数: $($parser.Methods.Count)" -ForegroundColor Gray
Write-Host "  グローバル変数数: $($parser.GlobalVars.Count)" -ForegroundColor Gray
Write-Host "  フォーム数: $($parser.Forms.Count)" -ForegroundColor Gray
Write-Host "  分岐数: $($parser.Branches.Count)" -ForegroundColor Gray

$report = [TextReport]::new($parser)
$text = $report.Generate()

if ($OutFile) {
    $text | Out-File -FilePath $OutFile -Encoding utf8
    Write-Host "レポート出力: $(Resolve-Path $OutFile)" -ForegroundColor Green
}
else {
    Write-Output $text
}

Write-Host '完了' -ForegroundColor Green
